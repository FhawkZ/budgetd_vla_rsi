"""Fetch Qwen's official ModelScope release and verify every artifact SHA-256."""
import concurrent.futures, hashlib, json, os, time, urllib.request, urllib.parse
from pathlib import Path
R=Path(os.environ['DEX_RUNTIME_ROOT']); O=R/'outputs/agent-rsi-20261024'; M=R/'models/agent-rsi/Qwen3-VL-8B-Instruct'; M.mkdir(parents=True,exist_ok=True)
def put(p,d):
 p=Path(p);q=p.with_suffix('.tmp');q.write_text(json.dumps(d,indent=2)+'\n');q.replace(p)
def digest(p):
 h=hashlib.sha256()
 with open(p,'rb') as f:
  for b in iter(lambda:f.read(8<<20),b''):h.update(b)
 return h.hexdigest()
meta=O/'qwen-modelscope-manifest.json'
if meta.exists(): files=json.loads(meta.read_text())['files']
else:
 u='https://modelscope.cn/api/v1/models/Qwen/Qwen3-VL-8B-Instruct/repo/files?Revision=master&Recursive=true'
 d=json.load(urllib.request.urlopen(u,timeout=30));assert d['Success']
 files=[f for f in d['Data']['Files'] if f['Type']=='blob' and (f['Path'].endswith(('.json','.safetensors','.txt','.jinja')) or f['Path']=='README.md')]
 assert any(f['Path'].endswith('.safetensors') for f in files)
 put(meta,{'repository':'Qwen/Qwen3-VL-8B-Instruct','source':'official ModelScope Qwen organization','fetched_unix':time.time(),'files':files})
print('total_bytes',sum(f['Size'] for f in files),flush=True)
def download(f):
 target=M/f['Path']; assert target.is_relative_to(M);target.parent.mkdir(parents=True,exist_ok=True)
 expected=f['Sha256'];assert len(expected)==64
 if target.exists() and target.stat().st_size==f['Size'] and digest(target)==expected:return {'path':f['Path'],'sha256':expected,'bytes':f['Size'],'reused':True}
 url='https://modelscope.cn/models/Qwen/Qwen3-VL-8B-Instruct/resolve/'+urllib.parse.quote(f['Revision'])+'/'+urllib.parse.quote(f['Path'])
 temp=target.with_suffix(target.suffix+'.partial')
 for attempt in range(3):
  try:
   offset=temp.stat().st_size if temp.exists() else 0
   req=urllib.request.Request(url,headers={'Range':f'bytes={offset}-'} if offset else {})
   with urllib.request.urlopen(req,timeout=90) as response:
    status=response.status
    if offset and status==206:assert response.headers.get('Content-Range','').startswith(f'bytes {offset}-')
    mode='ab' if offset and status==206 else 'wb'
    with open(temp,mode) as out:
     for block in iter(lambda:response.read(4<<20),b''):out.write(block)
   assert temp.stat().st_size==f['Size'],(f['Path'],temp.stat().st_size,f['Size'])
   assert digest(temp)==expected,('sha mismatch',f['Path'])
   temp.replace(target); print('verified',f['Path'],f['Size'],flush=True)
   return {'path':f['Path'],'sha256':expected,'bytes':f['Size'],'revision':f['Revision']}
  except Exception as e:
   print('attempt_failed',f['Path'],attempt+1,repr(e),flush=True)
   if temp.exists() and temp.stat().st_size>=f['Size']:temp.rename(temp.with_suffix(temp.suffix+f'.invalid{attempt}'))
   if attempt==2:raise
   time.sleep(2)
try:
 with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool: rows=list(pool.map(download,files))
 index=json.loads((M/'model.safetensors.index.json').read_text());assert set(index['weight_map'].values()).issubset({r['path'] for r in rows})
 put(O/'qwen-download-complete.json',{'status':'complete','model_path':str(M),'files':rows,'total_bytes':sum(r['bytes'] for r in rows),'finished_unix':time.time()})
except Exception as e:
 put(O/'qwen-download-error.json',{'error':repr(e),'time':time.time()});raise
